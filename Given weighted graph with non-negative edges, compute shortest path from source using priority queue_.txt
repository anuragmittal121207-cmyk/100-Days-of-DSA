#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define MAX 100

typedef struct {
    int node;
    int dist;
} HeapNode;

HeapNode heap[MAX];
int size = 0;

// Swap
void swap(int i, int j) {
    HeapNode temp = heap[i];
    heap[i] = heap[j];
    heap[j] = temp;
}

// Insert into min-heap
void push(int node, int dist) {
    int i = size++;
    heap[i].node = node;
    heap[i].dist = dist;

    while (i > 0 && heap[(i - 1)/2].dist > heap[i].dist) {
        swap(i, (i - 1)/2);
        i = (i - 1)/2;
    }
}

// Remove min
HeapNode pop() {
    HeapNode root = heap[0];
    heap[0] = heap[--size];

    int i = 0;
    while (2*i + 1 < size) {
        int left = 2*i + 1;
        int right = 2*i + 2;
        int smallest = left;

        if (right < size && heap[right].dist < heap[left].dist)
            smallest = right;

        if (heap[i].dist <= heap[smallest].dist)
            break;

        swap(i, smallest);
        i = smallest;
    }

    return root;
}

int isEmpty() {
    return size == 0;
}

void dijkstra(int V, int adj[MAX][MAX], int source) {
    int dist[MAX];
    int visited[MAX] = {0};

    for (int i = 0; i < V; i++)
        dist[i] = INT_MAX;

    dist[source] = 0;
    push(source, 0);

    while (!isEmpty()) {
        HeapNode top = pop();
        int u = top.node;

        if (visited[u]) continue;
        visited[u] = 1;

        for (int v = 0; v < V; v++) {
            if (adj[u][v] > 0 && !visited[v]) {
                int newDist = dist[u] + adj[u][v];

                if (newDist < dist[v]) {
                    dist[v] = newDist;
                    push(v, newDist);
                }
            }
        }
    }

    printf("Shortest distances from source %d:\n", source);
    for (int i = 0; i < V; i++) {
        printf("To %d = %d\n", i, dist[i]);
    }
}

int main() {
    int V = 5;

    int adj[MAX][MAX] = {
        {0, 10, 0, 5, 0},
        {0, 0, 1, 2, 0},
        {0, 0, 0, 0, 4},
        {0, 3, 9, 0, 2},
        {7, 0, 6, 0, 0}
    };

    dijkstra(V, adj, 0);

    return 0;
}
