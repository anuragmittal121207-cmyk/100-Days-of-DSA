#include <stdio.h>
#include <stdlib.h>

// Stack structure
typedef struct {
    int *arr;
    int top;
    int capacity;
} Stack;

// Initialize stack
void initStack(Stack *s, int capacity) {
    s->capacity = capacity;
    s->top = -1;
    s->arr = (int*)malloc(capacity * sizeof(int));
}

// Push
void push(Stack *s, int x) {
    s->arr[++s->top] = x;
}

// Pop
int pop(Stack *s) {
    return s->arr[s->top--];
}

// Queue using array
typedef struct {
    int *arr;
    int front, rear, size, capacity;
} Queue;

// Initialize queue
void initQueue(Queue *q, int capacity) {
    q->capacity = capacity;
    q->front = 0;
    q->size = 0;
    q->rear = capacity - 1;
    q->arr = (int*)malloc(capacity * sizeof(int));
}

// Enqueue
void enqueue(Queue *q, int x) {
    q->rear = (q->rear + 1) % q->capacity;
    q->arr[q->rear] = x;
    q->size++;
}

// Dequeue
int dequeue(Queue *q) {
    int x = q->arr[q->front];
    q->front = (q->front + 1) % q->capacity;
    q->size--;
    return x;
}

// Reverse Queue
void reverseQueue(Queue *q) {
    Stack s;
    initStack(&s, q->capacity);

    // Step 1: Move queue → stack
    while (q->size > 0) {
        push(&s, dequeue(q));
    }

    // Step 2: Move stack → queue
    while (s.top != -1) {
        enqueue(q, pop(&s));
    }

    free(s.arr);
}

// Main
int main() {
    int N;
    scanf("%d", &N);

    Queue q;
    initQueue(&q, N);

    for (int i = 0; i < N; i++) {
        int x;
        scanf("%d", &x);
        enqueue(&q, x);
    }

    reverseQueue(&q);

    // Print reversed queue
    while (q.size > 0) {
        printf("%d ", dequeue(&q));
    }

    free(q.arr);
    return 0;
}
