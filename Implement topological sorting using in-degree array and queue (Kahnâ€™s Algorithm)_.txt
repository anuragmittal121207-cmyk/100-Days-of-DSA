#include <stdio.h>
#include <stdlib.h>

#define MAX 100

int queue[MAX], front = 0, rear = -1;

// Queue operations
void enqueue(int x) {
    queue[++rear] = x;
}

int dequeue() {
    return queue[front++];
}

int isEmpty() {
    return front > rear;
}

void topologicalSort(int V, int adj[MAX][MAX]) {
    int in_degree[MAX] = {0};
    int topo[MAX], index = 0;

    // Step 1: Calculate in-degrees
    for (int i = 0; i < V; i++) {
        for (int j = 0; j < V; j++) {
            if (adj[i][j] == 1) {
                in_degree[j]++;
            }
        }
    }

    // Step 2: Add nodes with in-degree 0 to queue
    for (int i = 0; i < V; i++) {
        if (in_degree[i] == 0) {
            enqueue(i);
        }
    }

    // Step 3: Process queue
    while (!isEmpty()) {
        int node = dequeue();
        topo[index++] = node;

        for (int i = 0; i < V; i++) {
            if (adj[node][i] == 1) {
                in_degree[i]--;
                if (in_degree[i] == 0) {
                    enqueue(i);
                }
            }
        }
    }

    // Step 4: Check for cycle
    if (index != V) {
        printf("Cycle detected! Topological sort not possible.\n");
        return;
    }

    // Print result
    printf("Topological Order: ");
    for (int i = 0; i < V; i++) {
        printf("%d ", topo[i]);
    }
    printf("\n");
}

int main() {
    int V = 6;

    // Adjacency matrix representation
    int adj[MAX][MAX] = {0};

    // Edges: (u → v)
    adj[5][2] = 1;
    adj[5][0] = 1;
    adj[4][0] = 1;
    adj[4][1] = 1;
    adj[2][3] = 1;
    adj[3][1] = 1;

    topologicalSort(V, adj);

    return 0;
}
