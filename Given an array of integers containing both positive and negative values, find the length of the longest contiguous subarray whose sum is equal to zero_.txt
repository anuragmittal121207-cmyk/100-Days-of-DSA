#include <stdio.h>

#define MAX 10000

// Simple hash map using array (for prefix sums range handling)
typedef struct {
    int sum;
    int index;
} Hash;

int maxLen(int arr[], int n) {
    int prefix = 0, max_len = 0;

    // Use array as hashmap (offset for negatives)
    int hash[2 * MAX];
    for (int i = 0; i < 2 * MAX; i++)
        hash[i] = -2;  // -2 means unused

    // Offset to handle negative sums
    int offset = MAX;

    hash[offset] = -1;  // prefix sum 0 at index -1

    for (int i = 0; i < n; i++) {
        prefix += arr[i];

        int key = prefix + offset;

        if (hash[key] != -2) {
            int length = i - hash[key];
            if (length > max_len)
                max_len = length;
        } else {
            hash[key] = i;  // store first occurrence
        }
    }

    return max_len;
}

int main() {
    int arr[] = {15, -2, 2, -8, 1, 7, 10, 23};
    int n = sizeof(arr) / sizeof(arr[0]);

    printf("Longest subarray length: %d\n", maxLen(arr, n));

    return 0;
}
