#include <stdio.h>
#include <stdlib.h>
#include <limits.h>

#define MAX 1000

// -------- Graph Structures --------
typedef struct {
    int to, weight;
} Edge;

typedef struct {
    Edge* edges;
    int size;
} AdjList;

// -------- Min Heap Structures --------
typedef struct {
    int node, dist;
} HeapNode;

HeapNode heap[MAX * MAX];
int heapSize = 0;

// Swap
void swap(HeapNode* a, HeapNode* b) {
    HeapNode temp = *a;
    *a = *b;
    *b = temp;
}

// Push into heap
void push(int node, int dist) {
    heap[++heapSize] = (HeapNode){node, dist};
    int i = heapSize;

    while (i > 1 && heap[i].dist < heap[i / 2].dist) {
        swap(&heap[i], &heap[i / 2]);
        i /= 2;
    }
}

// Pop min
HeapNode pop() {
    HeapNode minNode = heap[1];
    heap[1] = heap[heapSize--];

    int i = 1;
    while (1) {
        int left = 2 * i, right = 2 * i + 1, smallest = i;

        if (left <= heapSize && heap[left].dist < heap[smallest].dist)
            smallest = left;
        if (right <= heapSize && heap[right].dist < heap[smallest].dist)
            smallest = right;

        if (smallest == i) break;

        swap(&heap[i], &heap[smallest]);
        i = smallest;
    }

    return minNode;
}

// -------- Dijkstra --------
void dijkstra(int V, AdjList* graph, int src) {
    int dist[MAX];

    // Initialize
    for (int i = 0; i < V; i++)
        dist[i] = INT_MAX;

    dist[src] = 0;
    push(src, 0);

    while (heapSize > 0) {
        HeapNode curr = pop();
        int u = curr.node;
        int d = curr.dist;

        if (d > dist[u]) continue;

        for (int i = 0; i < graph[u].size; i++) {
            int v = graph[u].edges[i].to;
            int w = graph[u].edges[i].weight;

            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                push(v, dist[v]);
            }
        }
    }

    // Print result
    printf("Shortest distances from source %d:\n", src);
    for (int i = 0; i < V; i++) {
        if (dist[i] == INT_MAX)
            printf("Node %d: INF\n", i);
        else
            printf("Node %d: %d\n", i, dist[i]);
    }
}

// -------- Main --------
int main() {
    int V = 5;

    AdjList graph[MAX];

    // Allocate adjacency lists
    for (int i = 0; i < V; i++) {
        graph[i].edges = malloc(10 * sizeof(Edge));
        graph[i].size = 0;
    }

    // Undirected graph (add both directions)
    graph[0].edges[graph[0].size++] = (Edge){1, 10};
    graph[1].edges[graph[1].size++] = (Edge){0, 10};

    graph[0].edges[graph[0].size++] = (Edge){4, 5};
    graph[4].edges[graph[4].size++] = (Edge){0, 5};

    graph[1].edges[graph[1].size++] = (Edge){2, 1};
    graph[2].edges[graph[2].size++] = (Edge){1, 1};

    graph[4].edges[graph[4].size++] = (Edge){1, 3};
    graph[1].edges[graph[1].size++] = (Edge){4, 3};

    graph[4].edges[graph[4].size++] = (Edge){2, 9};
    graph[2].edges[graph[2].size++] = (Edge){4, 9};

    graph[4].edges[graph[4].size++] = (Edge){3, 2};
    graph[3].edges[graph[3].size++] = (Edge){4, 2};

    graph[2].edges[graph[2].size++] = (Edge){3, 4};
    graph[3].edges[graph[3].size++] = (Edge){2, 4};

    dijkstra(V, graph, 0);

    return 0;
}
